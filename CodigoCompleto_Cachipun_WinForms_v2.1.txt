using System;

namespace Cachipun;

/// <summary>
/// Las tres jugadas. El orden importa: cada jugada le gana a la que va antes en la lista,
/// y Piedra (la primera) le gana a Tijera (la última). Papel > Piedra > Tijera > Papel.
/// </summary>
public enum Jugada
{
    Piedra = 0,
    Papel = 1,
    Tijera = 2
}

/// <summary>Cómo terminó una ronda, mirado desde el jugador.</summary>
public enum Resultado
{
    Ganaste,
    Perdiste,
    Empate
}

/// <summary>Los dos modos de la app: jugar contra la compu o pedir una jugada al azar.</summary>
public enum ModoJuego
{
    ContraLaCompu,
    Azar
}

/// <summary>Niveles de la compu. El número coincide con la posición en el ComboBox.</summary>
public enum Dificultad
{
    Facil = 0,
    Normal = 1,
    Dificil = 2
}

/// <summary>
/// Métodos de ayuda para las jugadas: nombres con su artículo correcto (la piedra, el papel,
/// la tijera), plurales, emojis y las reglas del juego. Las frases chilenas usan esto para no
/// decir cosas como "el piedra" o "dos papel seguidas".
/// </summary>
public static class JugadaExtensiones
{
    public static readonly Jugada[] Todas = { Jugada.Piedra, Jugada.Papel, Jugada.Tijera };

    /// <summary>"piedra", "papel", "tijera" (en minúscula, para meter en frases).</summary>
    public static string Nombre(this Jugada j)
    {
        switch (j)
        {
            case Jugada.Piedra: return "piedra";
            case Jugada.Papel: return "papel";
            default: return "tijera";
        }
    }

    /// <summary>"Piedra", "Papel", "Tijera" (para títulos y botones).</summary>
    public static string NombreTitulo(this Jugada j)
    {
        string n = j.Nombre();
        return char.ToUpper(n[0]) + n.Substring(1);
    }

    /// <summary>"la" para piedra y tijera, "el" para papel.</summary>
    public static string Articulo(this Jugada j)
    {
        return j == Jugada.Papel ? "el" : "la";
    }

    /// <summary>"la piedra", "el papel", "la tijera".</summary>
    public static string ConArticulo(this Jugada j)
    {
        return j.Articulo() + " " + j.Nombre();
    }

    /// <summary>"piedras", "papeles", "tijeras".</summary>
    public static string Plural(this Jugada j)
    {
        switch (j)
        {
            case Jugada.Piedra: return "piedras";
            case Jugada.Papel: return "papeles";
            default: return "tijeras";
        }
    }

    /// <summary>"esa piedra", "ese papel", "esa tijera".</summary>
    public static string Esa(this Jugada j)
    {
        return j == Jugada.Papel ? "ese papel" : "esa " + j.Nombre();
    }

    /// <summary>Emoji de la mano. En WinForms salen en blanco y negro y toman el color del texto.</summary>
    public static string Emoji(this Jugada j)
    {
        switch (j)
        {
            case Jugada.Piedra: return "✊";
            case Jugada.Papel: return "✋";
            default: return "✌";
        }
    }

    /// <summary>La jugada que le gana a esta (Piedra → Papel, Papel → Tijera, Tijera → Piedra).</summary>
    public static Jugada QuienLeGana(this Jugada j)
    {
        return (Jugada)(((int)j + 1) % 3);
    }

    /// <summary>La jugada a la que esta le gana (Piedra → Tijera, Papel → Piedra, Tijera → Papel).</summary>
    public static Jugada AQuienLeGana(this Jugada j)
    {
        return (Jugada)(((int)j + 2) % 3);
    }

    /// <summary>Resultado de tirar "tuya" contra "otra", mirado desde quien tiró "tuya".</summary>
    public static Resultado ContraOtra(this Jugada tuya, Jugada otra)
    {
        if (tuya == otra) return Resultado.Empate;
        return otra == tuya.AQuienLeGana() ? Resultado.Ganaste : Resultado.Perdiste;
    }

    /// <summary>La regla completa de la jugada ganadora, como en la web de referencia ("Papel cubre a Piedra").</summary>
    public static string Regla(this Jugada ganadora)
    {
        switch (ganadora)
        {
            case Jugada.Piedra: return "La piedra aplasta a la tijera";
            case Jugada.Papel: return "El papel envuelve a la piedra";
            default: return "La tijera corta el papel";
        }
    }

    /// <summary>Versión corta para los botones.</summary>
    public static string ReglaCorta(this Jugada j)
    {
        switch (j)
        {
            case Jugada.Piedra: return "le gana a la tijera";
            case Jugada.Papel: return "le gana a la piedra";
            default: return "le gana al papel";
        }
    }
}

using System;
using System.Collections.Generic;
using System.Linq;

namespace Cachipun;

/// <summary>
/// Lleva la cuenta de la partida: victorias, derrotas, empates y rachas.
/// Regla de la casa: un empate no suma ni corta la racha.
/// </summary>
public sealed class Estadisticas
{
    public int Victorias { get; private set; }
    public int Derrotas { get; private set; }
    public int Empates { get; private set; }

    /// <summary>Victorias seguidas ahora mismo.</summary>
    public int RachaActual { get; private set; }

    /// <summary>La racha de victorias más larga de la sesión.</summary>
    public int MejorRacha { get; private set; }

    /// <summary>Derrotas seguidas ahora mismo.</summary>
    public int RachaDerrotas { get; private set; }

    public int EmpatesSeguidos { get; private set; }

    /// <summary>Resultado de la ronda anterior (null si todavía no se juega ninguna).</summary>
    public Resultado? Anterior { get; private set; }

    /// <summary>Se pone en true cuando la última ronda batió la mejor racha (desde 3 victorias seguidas).</summary>
    public bool BatioRecord { get; private set; }

    public int Rondas
    {
        get { return Victorias + Derrotas + Empates; }
    }

    /// <summary>Porcentaje de rondas ganadas (0 a 100). Los empates cuentan como ronda jugada.</summary>
    public double TasaExito
    {
        get { return Rondas == 0 ? 0 : Victorias * 100.0 / Rondas; }
    }

    public void Registrar(Resultado resultado)
    {
        BatioRecord = false;

        switch (resultado)
        {
            case Resultado.Ganaste:
                Victorias++;
                RachaActual++;
                RachaDerrotas = 0;
                EmpatesSeguidos = 0;
                BatioRecord = RachaActual > MejorRacha && RachaActual >= 3;
                MejorRacha = Math.Max(MejorRacha, RachaActual);
                break;

            case Resultado.Perdiste:
                Derrotas++;
                RachaDerrotas++;
                RachaActual = 0;
                EmpatesSeguidos = 0;
                break;

            default:
                Empates++;
                EmpatesSeguidos++;
                break;
        }

        Anterior = resultado;
    }

    public void Reiniciar()
    {
        Victorias = Derrotas = Empates = 0;
        RachaActual = MejorRacha = RachaDerrotas = EmpatesSeguidos = 0;
        Anterior = null;
        BatioRecord = false;
    }
}

/// <summary>
/// El "cerebro" de la compu. No sabe nada de ventanas ni botones, solo de jugadas.
///
/// Importante: la compu decide ANTES de enterarse de tu jugada de esta ronda. Por eso
/// <see cref="ElegirJugadaCompu"/> no recibe tu jugada, y después hay que llamar a
/// <see cref="RegistrarJugadaDelJugador"/>. Así no hace trampa.
/// </summary>
public sealed class MotorCachipun
{
    private readonly Random azar;
    private readonly int[] frecuencia = new int[3];              // cuántas veces has tirado cada jugada
    private readonly int[,] transiciones = new int[3, 3];        // [jugada anterior, jugada siguiente]
    private Jugada? ultima;

    public MotorCachipun(Random? azar = null)
    {
        this.azar = azar ?? new Random();
    }

    public Jugada JugadaAlAzar()
    {
        return (Jugada)azar.Next(3);
    }

    public Jugada ElegirJugadaCompu(Dificultad nivel)
    {
        switch (nivel)
        {
            case Dificultad.Facil:
                return JugadaAlAzar();

            case Dificultad.Normal:
                // 4 de cada 10 veces le gana a tu última jugada; el resto, al azar.
                if (ultima is Jugada anterior && azar.NextDouble() < 0.40)
                    return anterior.QuienLeGana();
                return JugadaAlAzar();

            default:
                // Difícil: intenta adivinar qué vas a tirar y le gana (7 de cada 10 veces).
                Jugada? prediccion = PredecirJugadaDelJugador();
                if (prediccion is Jugada esperada && azar.NextDouble() < 0.70)
                    return esperada.QuienLeGana();
                return JugadaAlAzar();
        }
    }

    public void RegistrarJugadaDelJugador(Jugada jugada)
    {
        if (ultima is Jugada anterior)
            transiciones[(int)anterior, (int)jugada]++;

        frecuencia[(int)jugada]++;
        ultima = jugada;
    }

    public void Reiniciar()
    {
        Array.Clear(frecuencia, 0, frecuencia.Length);
        Array.Clear(transiciones, 0, transiciones.Length);
        ultima = null;
    }

    /// <summary>
    /// Primero mira qué sueles tirar DESPUÉS de tu última jugada (por ejemplo, "después de piedra
    /// casi siempre tira papel"). Si todavía hay pocos datos, usa la jugada que más has tirado.
    /// </summary>
    private Jugada? PredecirJugadaDelJugador()
    {
        if (ultima is Jugada anterior)
        {
            int[] fila =
            {
                transiciones[(int)anterior, 0],
                transiciones[(int)anterior, 1],
                transiciones[(int)anterior, 2]
            };

            if (fila.Sum() >= 3)
                return (Jugada)IndiceMayor(fila);
        }

        if (frecuencia.Sum() >= 2)
            return (Jugada)IndiceMayor(frecuencia);

        return null;
    }

    /// <summary>Índice del valor más grande; si hay empate entre varios, elige uno al azar.</summary>
    private int IndiceMayor(int[] valores)
    {
        int maximo = valores.Max();
        var empatados = new List<int>();

        for (int i = 0; i < valores.Length; i++)
        {
            if (valores[i] == maximo) empatados.Add(i);
        }

        return empatados[azar.Next(empatados.Count)];
    }
}

/// <summary>
/// Banco de frases chilenas. Cada frase es una plantilla con "marcas" entre llaves que se
/// reemplazan según las jugadas de la ronda, así el género y el plural siempre quedan bien:
///
///   {tu} / {Tu}          → "tu piedra" / "Tu piedra"        (tu jugada)
///   {mi} / {Mi}          → "mi tijera" / "Mi tijera"        (jugada de la compu)
///   {laTuya} / {LaTuya}  → "la piedra" / "el papel"         (tu jugada con artículo)
///   {laMia} / {LaMia}    → "la tijera" / "el papel"         (jugada de la compu con artículo)
///   {jugadaTu}           → "piedra"                         (tu jugada, sola)
///   {jugadaMia}          → "tijera"                         (jugada de la compu, sola)
///   {plural}             → "piedras" / "papeles" / "tijeras"
///   {esa}                → "esa piedra" / "ese papel"
///   {n}                  → un número (racha, repeticiones, etc.)
///
/// Las frases evitan adjetivos con género sobre quien juega ("cómodo", "cachado"), para que
/// funcionen igual si juega un hombre o una mujer.
/// </summary>
public sealed class FrasesChilenas
{
    private readonly Random azar;
    private readonly Dictionary<string, string> ultimaPorCategoria = new Dictionary<string, string>();

    public FrasesChilenas(Random azar)
    {
        this.azar = azar;
    }

    /// <summary>Todas las plantillas (se deja público para poder revisarlas en pruebas).</summary>
    public static IReadOnlyDictionary<string, string[]> Plantillas
    {
        get { return Banco; }
    }

    /// <summary>
    /// Elige una frase de la categoría, sin repetir la misma dos veces seguidas,
    /// y reemplaza las marcas con las jugadas indicadas.
    /// </summary>
    public string Elegir(string categoria, Jugada? tuya = null, Jugada? compu = null, int n = 0)
    {
        if (!Banco.TryGetValue(categoria, out string[]? opciones) || opciones.Length == 0)
            return string.Empty;

        string plantilla = opciones[0];

        if (opciones.Length > 1)
        {
            ultimaPorCategoria.TryGetValue(categoria, out string? anterior);
            string[] candidatas = anterior == null
                ? opciones
                : opciones.Where(o => o != anterior).ToArray();

            if (candidatas.Length == 0) candidatas = opciones;

            plantilla = candidatas[azar.Next(candidatas.Length)];
            ultimaPorCategoria[categoria] = plantilla;
        }

        return Armar(plantilla, tuya, compu, n);
    }

    /// <summary>Frase especial cuando se llega a cierta cantidad de rondas (null si no toca ninguna).</summary>
    public string? Hito(int rondas)
    {
        return Hitos.TryGetValue(rondas, out string? frase) ? frase : null;
    }

    public void Reiniciar()
    {
        ultimaPorCategoria.Clear();
    }

    private static string Armar(string plantilla, Jugada? tuya, Jugada? compu, int n)
    {
        string r = plantilla;

        if (tuya is Jugada t)
        {
            r = r.Replace("{Tu}", "Tu " + t.Nombre())
                 .Replace("{tu}", "tu " + t.Nombre())
                 .Replace("{LaTuya}", Mayuscula(t.ConArticulo()))
                 .Replace("{laTuya}", t.ConArticulo())
                 .Replace("{jugadaTu}", t.Nombre())
                 .Replace("{plural}", t.Plural())
                 .Replace("{esa}", t.Esa());
        }

        if (compu is Jugada c)
        {
            r = r.Replace("{Mi}", "Mi " + c.Nombre())
                 .Replace("{mi}", "mi " + c.Nombre())
                 .Replace("{LaMia}", Mayuscula(c.ConArticulo()))
                 .Replace("{laMia}", c.ConArticulo())
                 .Replace("{jugadaMia}", c.Nombre());
        }

        return r.Replace("{n}", n.ToString());
    }

    private static string Mayuscula(string texto)
    {
        return char.ToUpper(texto[0]) + texto.Substring(1);
    }

/// <summary>Colores de la app, todos en un solo lugar para cambiarlos fácil.</summary>
internal static class Tema
{
    public static readonly Color Fondo = Color.FromArgb(247, 248, 252);
    public static readonly Color FondoTarjeta = Color.White;
    public static readonly Color FondoDuelo = Color.FromArgb(248, 247, 255);

    public static readonly Color Neutro = Color.FromArgb(245, 246, 250);

    public static readonly Color Tinta = Color.FromArgb(39, 43, 61);
    public static readonly Color Suave = Color.FromArgb(112, 119, 139);
    public static readonly Color Morado = Color.FromArgb(112, 86, 235);
    public static readonly Color MoradoClaro = Color.FromArgb(241, 237, 255);
    public static readonly Color Verde = Color.FromArgb(44, 154, 95);
    public static readonly Color Rojo = Color.FromArgb(219, 77, 91);
    public static readonly Color Dorado = Color.FromArgb(194, 139, 28);

    /// <summary>Mezcla dos colores: t = 0 devuelve "a", t = 1 devuelve "b".</summary>
    public static Color Mezclar(Color a, Color b, double t)
    {
        t = Math.Max(0, Math.Min(1, t));
        return Color.FromArgb(
            (int)(a.R + (b.R - a.R) * t),
            (int)(a.G + (b.G - a.G) * t),
            (int)(a.B + (b.B - a.B) * t));
    }
}

/// <summary>
/// Panel blanco con borde fino y un poco de espacio interior: la "tarjeta" que usa toda la interfaz.
/// Se estira solo para llenar la celda donde se pone.
/// </summary>
internal sealed class Tarjeta : Panel
{
    public Tarjeta()
    {
        DoubleBuffered = true;
        ResizeRedraw = true;
        BackColor = Tema.FondoTarjeta;
        Padding = new Padding(16, 12, 16, 12);
        Margin = new Padding(0);
        Dock = DockStyle.Fill;
    }

    protected override void OnPaint(PaintEventArgs e)
    {
        base.OnPaint(e);
        using (var lapiz = new Pen(Tema.Borde))
        {
            e.Graphics.DrawRectangle(lapiz, 0, 0, Width - 1, Height - 1);
        }
    }
}

/// <summary>Un renglón del historial: el texto y el color con que se dibuja.</summary>
internal sealed class RegistroHistorial
{
    public RegistroHistorial(string texto, Color color)
    {
        Texto = texto;
        Color = color;
    }

    public string Texto { get; }
    public Color Color { get; }

    public override string ToString()
    {
        return Texto;
    }
}

    private bool ModoAzar
    {
        get { return rbAzar.Checked; }
    }

    private void Tirar(Jugada tuya)
    {
        if (jugando || ModoAzar) return;

        jugadaPendiente = tuya;
        Bloquear(true);
        ComenzarCuentaORevelar();
    }

    private void PedirJugadaAlAzar()
    {
        if (jugando || !ModoAzar) return;

        jugadaPendiente = motor.JugadaAlAzar();
        Bloquear(true);
        ComenzarCuentaORevelar();
    }

    private void ComenzarCuentaORevelar()
    {
        if (chkCuenta.Checked)
        {
            pasoCuenta = 0;
            MostrarPasoDeLaCuenta();
            timerCuenta.Start();
        }
        else
        {
            Revelar();
        }
    }

    private void AlTickCuenta(object? sender, EventArgs e)
    {
        pasoCuenta++;

        if (pasoCuenta >= PalabrasCuenta.Length)
        {
            timerCuenta.Stop();
            Revelar();
            return;
        }

        MostrarPasoDeLaCuenta();
    }

    /// <summary>"¡Ca…" "chi…" "¡PÚN!": los puños suben y bajan (o giran los emojis, en modo azar).</summary>
    private void MostrarPasoDeLaCuenta()
    {
        lblTitulo.Text = PalabrasCuenta[pasoCuenta];
        lblTitulo.ForeColor = Tema.Morado;
        lblRegla.Text = string.Empty;
        lblDetalle.Text = string.Empty;
        acentoResultado.BackColor = Tema.Morado;
        contenidoResultado.BackColor = Tema.Neutro;

        if (ModoAzar)
        {
            lblIconoTu.Text = JugadaExtensiones.Todas[azar.Next(3)].Emoji();
            lblIconoCompu.Text = "🎲";
        }
        else
        {
            lblIconoTu.Text = Jugada.Piedra.Emoji();
            lblIconoCompu.Text = Jugada.Piedra.Emoji();
        }

        lblNombreTu.Text = "…";
        lblNombreCompu.Text = "…";

        // El puño "late": chico en "¡Ca…" y "¡PÚN!", grande en "chi…".
        Font fuente = pasoCuenta % 2 == 0 ? fuenteIconoChica : fuenteIcono;
        lblIconoTu.Font = fuente;
        lblIconoCompu.Font = fuente;
    }

    private void Revelar()
    {
        lblIconoTu.Font = fuenteIcono;
        lblIconoCompu.Font = fuenteIcono;

        if (ModoAzar)
            RevelarJugadaAlAzar();
        else
            RevelarRondaContraLaCompu();

        Bloquear(false);
    }

    private void RevelarRondaContraLaCompu()
    {
        Jugada tuya = jugadaPendiente;
        Resultado? anterior = estadisticas.Anterior;

        // La compu decide ANTES de registrar tu jugada de esta ronda: así no hace trampa.
        Jugada compu = motor.ElegirJugadaCompu(NivelActual());
        motor.RegistrarJugadaDelJugador(tuya);

        Resultado resultado = tuya.ContraOtra(compu);
        estadisticas.Registrar(resultado);

        repeticiones = ultimaTuya == tuya ? repeticiones + 1 : 1;
        ultimaTuya = tuya;

        lblIconoTu.Text = tuya.Emoji();
        lblIconoCompu.Text = compu.Emoji();
        lblNombreTu.Text = tuya.NombreTitulo();
        lblNombreCompu.Text = compu.NombreTitulo();

        Color color = Tema.Suave;
        string titulo = string.Empty;
        string regla = string.Empty;
        string detalle = string.Empty;
        string estado = string.Empty;
        int racha = estadisticas.RachaActual;

        switch (resultado)
        {
            case Resultado.Ganaste:
                color = Tema.Verde;
                titulo = frases.Elegir(Categoria.TituloGana);
                regla = tuya.Regla();
                if (racha >= 3)
                {
                    detalle = frases.Elegir(Categoria.DetalleGanaRacha, tuya, compu, racha);
                    estado = frases.Elegir(Categoria.EstadoGanaRacha, tuya, compu, racha);
                }
                else
                {
                    detalle = anterior == Resultado.Perdiste
                        ? frases.Elegir(Categoria.DetalleGanaVuelta, tuya, compu)
                        : frases.Elegir(Categoria.DetalleGana, tuya, compu);
                    estado = frases.Elegir(Categoria.EstadoGana);
                }
                break;

            case Resultado.Perdiste:
                color = Tema.Rojo;
                titulo = frases.Elegir(Categoria.TituloPierde);
                regla = compu.Regla();
                if (estadisticas.RachaDerrotas >= 3)
                {
                    detalle = frases.Elegir(Categoria.DetallePierdeRacha, tuya, compu, estadisticas.RachaDerrotas);
                    estado = frases.Elegir(Categoria.EstadoPierdeRacha);
                }
                else
                {
                    detalle = anterior == Resultado.Ganaste
                        ? frases.Elegir(Categoria.DetallePierdeCortaRacha, tuya, compu)
                        : frases.Elegir(Categoria.DetallePierde, tuya, compu);
                    estado = frases.Elegir(Categoria.EstadoPierde);
                }
                break;

            default:
                color = Tema.Dorado;
                titulo = frases.Elegir(Categoria.TituloEmpate);
                regla = "Los dos tiramos " + tuya.Nombre();
                detalle = estadisticas.EmpatesSeguidos >= 2
                    ? frases.Elegir(Categoria.DetalleEmpateSeguido, tuya, compu, estadisticas.EmpatesSeguidos)
                    : frases.Elegir(Categoria.DetalleEmpate, tuya, compu);
                estado = frases.Elegir(Categoria.EstadoEmpate);
                break;
        }

        // Mensajes especiales para la barra de abajo: récord > hito de rondas > "repites mucho la misma".
        if (estadisticas.BatioRecord)
        {
            estado = frases.Elegir(Categoria.EstadoRecord, n: estadisticas.MejorRacha);
        }
        else
        {
            string? hito = frases.Hito(estadisticas.Rondas);
            if (hito != null)
                estado = hito;
            else if (repeticiones >= 2 && azar.NextDouble() < 0.40)
                estado = frases.Elegir(Categoria.EstadoRepite, tuya, compu, repeticiones);
        }

        MostrarResultado(titulo, regla, detalle, color);
        SetEstado(estado, color);
        AgregarAlHistorial(
            estadisticas.Rondas.ToString("00") + "  " + tuya.NombreTitulo() + " vs " + compu.NombreTitulo() + "  ·  " + resultado,
            color);
        ActualizarEstadisticas();
    }

    private void DibujarRenglonDelHistorial(object? sender, DrawItemEventArgs e)
    {
        if (e.Index < 0) return;

        e.DrawBackground();

        if (lstHistorial.Items[e.Index] is RegistroHistorial registro)
        {
            TextRenderer.DrawText(
                e.Graphics,
                registro.Texto,
                lstHistorial.Font,
                e.Bounds,
                registro.Color,
                TextFormatFlags.Left | TextFormatFlags.VerticalCenter | TextFormatFlags.EndEllipsis);
        }
    }

    private void AlCambiarNivel()
    {
        string categoria;
        switch (NivelActual())
        {

    }

        if (jugando) return;

        estadisticas.Reiniciar();
        motor.Reiniciar();
        frases.Reiniciar();
        ultimaTuya = null;
        repeticiones = 0;
        tiradasAzar = 0;

        timerDestello.Stop();
        lstHistorial.Items.Clear();
        MostrarPanelInicial();
        ActualizarEstadisticas();
        SetEstado(frases.Elegir(Categoria.EstadoReset), Tema.Suave);
    }

    private void AlPresionarTecla(object? sender, KeyEventArgs e)
    {
        if (e.Control && e.KeyCode == Keys.S)
        {
            SacarPantallazo();
            e.SuppressKeyPress = true;
            return;
        }

        if (e.Control && e.KeyCode == Keys.N)
        {
            PartirDeNuevo();
            e.SuppressKeyPress = true;
            return;
        }

        if (e.Control || e.Alt || ModoAzar) return;

        if (e.KeyCode == Keys.D1 || e.KeyCode == Keys.NumPad1) Tirar(Jugada.Piedra);
        else if (e.KeyCode == Keys.D2 || e.KeyCode == Keys.NumPad2) Tirar(Jugada.Papel);
        else if (e.KeyCode == Keys.D3 || e.KeyCode == Keys.NumPad3) Tirar(Jugada.Tijera);
        else return;

        e.SuppressKeyPress = true;
    }

    /// <summary>
    /// Guarda en PNG lo que se ve en la ventana. Se usa CopyFromScreen (y no DrawToBitmap) para que
    /// salga igual que en pantalla, sin la barra de título cortando el borde de abajo.
    /// </summary>
    private void SacarPantallazo()
    {
        try
        {
            string carpeta = Path.Combine(
                Environment.GetFolderPath(Environment.SpecialFolder.DesktopDirectory),
                "PantallazosCachipun");
            Directory.CreateDirectory(carpeta);

            string archivo = Path.Combine(carpeta, "cachipun_" + DateTime.Now.ToString("yyyyMMdd_HHmmss") + ".png");
            Point origen = PointToScreen(Point.Empty);

            using (var imagen = new Bitmap(ClientSize.Width, ClientSize.Height))
            {
                using (var g = Graphics.FromImage(imagen))
                {
                    g.CopyFromScreen(origen, Point.Empty, ClientSize);
                }

                imagen.Save(archivo, System.Drawing.Imaging.ImageFormat.Png);
            }

            SetEstado(frases.Elegir(Categoria.EstadoPantallazo), Tema.Verde);
        }
        catch (Exception ex)
        {
            MessageBox.Show(
                "Pucha, no pude guardar el pantallazo: " + ex.Message,
                "Pantallazo",
                MessageBoxButtons.OK,
                MessageBoxIcon.Warning);
        }
    }
}

# Cachipún — Windows Forms (versión 2)

Juego de piedra, papel o tijera (el cachipún de toda la vida) en **C# con Windows Forms** y **.NET 8**, para Windows.
La compu habla en chileno, lleva estadísticas de la sesión y tiene tres niveles de dificultad.

## Qué trae

- **Dos modos de juego**
  - *Jugar contra la compu*: eliges Piedra, Papel o Tijera y ves el resultado de la batalla (TÚ vs LA COMPU).
  - *Tírame una jugada al azar*: la compu sortea una jugada por ti (no cuenta para las estadísticas).
- **Cuenta regresiva «¡Ca… chi… PÚN!»** antes de revelar la jugada (se puede desactivar).
- **Tres niveles de la compu**
  - *Fácil*: tira cualquier cosa.
  - *Normal*: 4 de cada 10 veces le gana a tu última jugada.
  - *Difícil*: aprende qué sueles tirar después de cada jugada y trata de adivinar la siguiente.
  - En todos los niveles la compu decide **antes** de enterarse de tu jugada, así que no hace trampa.
- **Resultado con la regla**: «La piedra aplasta a la tijera», «El papel envuelve a la piedra», «La tijera corta el papel».
- **Estadísticas de la sesión**: victorias, derrotas, empates, racha actual, mejor racha y tasa de éxito.
  Un empate no suma ni corta la racha.
- **Historial** de las últimas 30 jugadas, con colores.
- **Más de 110 frases chilenas** (en 29 categorías) que cambian según lo que pasó: racha de victorias, vuelta después de perder,
  derrotas seguidas, empates seguidos, si repites mucho la misma jugada, nuevo récord, hitos de 10/25/50/100 rondas, etc.
  Nunca repite la misma frase dos veces seguidas y el género y el plural siempre quedan bien («la piedra», «el papel», «dos tijeras»).
- **Pantallazo** de la ventana en PNG, guardado en el Escritorio, carpeta `PantallazosCachipun`.

## Atajos

| Tecla | Qué hace |
|---|---|
| `1` / `2` / `3` | Piedra / Papel / Tijera |
| `Ctrl+S` | Sacar pantallazo |
| `Ctrl+N` | Partir de nuevo |

## Cómo abrirlo

1. Descomprime el ZIP.
2. Abre `Cachipun.csproj` en **Visual Studio 2022** (con la carga de trabajo *Desarrollo de escritorio con .NET*).
3. Aprieta **F5**.

También se puede correr desde una terminal, en la carpeta del proyecto: `dotnet run` (requiere el SDK de .NET 8 en Windows).

## Archivos

| Archivo | Contenido |
|---|---|
| `Program.cs` | Punto de entrada. |
| `Jugada.cs` | Enums (`Jugada`, `Resultado`, `ModoJuego`, `Dificultad`) y reglas del juego, con artículos y plurales correctos. |
| `MotorCachipun.cs` | `Estadisticas` (conteos y rachas) y `MotorCachipun` (la IA de la compu). |
| `FrasesChilenas.cs` | Banco de frases con marcas como `{tu}`, `{laMia}`, `{plural}`. |
| `Controles.cs` | Colores (`Tema`), la tarjeta blanca (`Tarjeta`) y el renglón del historial. |
| `MainForm.cs` | Interfaz y flujo del juego. |

La lógica (jugadas, IA, estadísticas, frases) no depende de Windows Forms, así que se puede probar por separado.

## Cómo agregar frases

Abre `FrasesChilenas.cs`, busca la categoría y agrega una línea. Las marcas disponibles:

| Marca | Resultado de ejemplo |
|---|---|
| `{tu}` / `{Tu}` | tu piedra / Tu piedra |
| `{mi}` / `{Mi}` | mi tijera / Mi tijera (la jugada de la compu) |
| `{laTuya}` / `{LaTuya}` | la piedra / el papel |
| `{laMia}` / `{LaMia}` | la tijera / el papel (compu) |
| `{jugadaTu}` / `{jugadaMia}` | piedra / tijera (sola, sin artículo) |
| `{plural}` | piedras / papeles / tijeras |
| `{esa}` | esa piedra / ese papel |
| `{n}` | un número (racha, repeticiones…) |

## Notas

- Los emojis en Windows Forms se dibujan en blanco y negro con el color del texto (es una limitación de WinForms); por eso van pintados de morado.
- Si la ventana se hace más chica que el tamaño mínimo, aparecen barras de desplazamiento en vez de cortar los botones.
